| 改造项 | 优先级 | 详细说明 |
|-------|-------|--------|

```python
def calculate_rcfd(graph, query_obj: str, neighbor_distance: float = 2.0):
    """
    计算关联置信度 R_cfd(query_obj | nearby_obj)
    
    逻辑：遍历图中所有时间戳，统计 query_obj 与其邻域物体的空间共现频率
    """
    cooccur_dict = {}  # {neighbor_obj: count}
    
    for obj in graph["floors"][0]["rooms"][0]["objects"]:
        if obj["category"] == query_obj:
            # 找到查询物体，统计其附近的物体
            query_pos = np.array(obj["pos_3d"])
            
            for nearby_obj in graph["floors"][0]["rooms"][0]["objects"]:
                if nearby_obj["category"] != query_obj:
                    nearby_pos = np.array(nearby_obj["pos_3d"])
                    dist = np.linalg.norm(query_pos - nearby_pos)
                    
                    if dist < neighbor_distance:
                        cooccur_dict[nearby_obj["obj_id"]] = \
                            cooccur_dict.get(nearby_obj["obj_id"], 0) + 1
    
    # 归一化
    total_count = sum(cooccur_dict.values()) + 1e-8
    rcfd = {k: v / total_count for k, v in cooccur_dict.items()}
    
    return rcfd
```

```python
        # 综合评分

    # Step 2: Agent 打分
    agent_priors = assign_spatial_priors(query_text, "living_room")
    
    # Step 3: 动态权重融合（随探索次数衰减 Agent 权重）
    w_agent = max(0.1, np.exp(-0.5 * n_explorations))  # 指数衰减
    w_rag = 1 - w_agent
    
    final_scores = {
        obj_id: w_rag * rag_scores[obj_id] + w_agent * agent_priors.get(obj_id, 0.1)
        for obj_id in rag_scores
    }
    
    return final_scores
```

```python
from sklearn.mixture import GaussianMixture
import matplotlib.pyplot as plt

def generate_gmm_heatmap(scores_dict, positions_dict, query_text: str):
    """
    输入：物体 ID -> 评分 + 物体 ID -> 3D 位置
    输出：2D 热力图 PDF，标注 Top-3 探索点 A/B/C
    """
    
    # 排序并取 Top-K
    sorted_objects = sorted(scores_dict.items(), key=lambda x: x[1], reverse=True)
    top_k = sorted_objects[:5]
    
    # 提取坐标（投影到 2D 平面）
    positions_2d = np.array([positions_dict[obj_id][:2] for obj_id, _ in top_k])
    scores = np.array([score for _, score in top_k])
    
    # 拟合高斯混合模型
    gmm = GaussianMixture(n_components=min(3, len(top_k)), random_state=42)
    gmm.fit(positions_2d, sample_weight=scores)
    
    # 生成网格热力图
    xx, yy = np.meshgrid(np.linspace(-1, 5, 100), np.linspace(-1, 5, 100))
    grid = np.c_[xx.ravel(), yy.ravel()]
    probs = np.exp(gmm.score_samples(grid)).reshape(xx.shape)
    
    # 绘图
    fig, ax = plt.subplots(figsize=(10, 8))
    contourf = ax.contourf(xx, yy, probs, levels=20, cmap="YlOrRd")
    
    # 标注 Top-3 点
    for i, (obj_id, score) in enumerate(top_k[:3]):
        pos = positions_dict[obj_id][:2]
        ax.plot(pos[0], pos[1], 'b*', markersize=20)
        ax.text(pos[0]+0.1, pos[1]+0.1, f"{'ABC'[i]}", fontsize=16, fontweight='bold')
    
    ax.set_title(f'GMM Heatmap for "{query_text}"', fontsize=14)
    ax.set_xlabel('X (meters)')
    ax.set_ylabel('Y (meters)')
    plt.colorbar(contourf, label="Probability")
    plt.savefig(f"heatmap_{query_text.replace(' ', '_')}.pdf", dpi=300)
    plt.close()
```

#### 4.5.1 理解失败经验的价值

**现象**：机器人在厨房桌子上寻找杯子，完整扫描后未找到。但系统仍然在下一轮搜索中反复建议"去桌子"。  
**解决**：引入"假阴性观测"的贝叶斯惩罚，系统应该记住"桌子已被验证不存在杯子"，并自动降权该位置。

**论文价值**：
- 证明你的系统从被动记忆升级到**主动遗忘**
- 对标 RobotMem 等长期导航工作，但用更简洁的数学表示
- 消融实验中，与"无记仇"对比，能显著提升成功率

```python
def query_with_negative_feedback(merged_graph, query_text, 
                                 failed_positions, n_explorations=1):
    """
    在 GMM 中融合正向和负向高斯核
    
    Args:
        failed_positions: 已验证失败的位置列表 [{pos: [x,y], confidence: 0.8}, ...]
    """
    
    # Step 1: 计算正向 GMM（和之前一样）
    final_scores = query_object(merged_graph, query_text, n_explorations)
    top_k_pos = extract_top_k_positions(final_scores, k=5)
    
    # Step 2: 为失败位置生成负向高斯核
    negative_kernels = []
    for failed_pos in failed_positions:
        # 失败位置的置信度越高，负向惩罚越强
        alpha_neg = 0.6 * failed_pos["confidence"]  # 范围 [0, 0.6]
        
        # 方差较大，表示"这片区域及周边都可能没有"
        sigma_neg = 1.5  # 比正向核更宽阔
        
        neg_kernel = Gaussian2D(
            mean=np.array(failed_pos["pos"]),
            cov=np.eye(2) * sigma_neg**2,
            weight=-alpha_neg  # 负权重
        )
        negative_kernels.append(neg_kernel)
    
    # Step 3: 合并正负高斯核
    gmm_grid = generate_gmm_probability_grid(top_k_pos)
    
    for neg_kernel in negative_kernels:
        neg_prob = neg_kernel.evaluate_grid(gmm_grid)
        gmm_grid -= neg_prob  # 注入负向
    
    # Step 4: 截断负值并重新归一化
    gmm_grid = np.maximum(gmm_grid, 0)  # 避免概率为负
    gmm_grid /= (np.sum(gmm_grid) + 1e-8)
    
    return gmm_grid, negative_kernels
```

```python
    failed_positions = [{"pos": [1.0, 0.5], "confidence": 0.85}]

    # 加入负向后：(1.0, 0.5) 的概率应下降 50% 以上
    
    gmm_normal = query_object(graph, "cup", n_explorations=1)
    prob_at_failed = gmm_normal.evaluate([1.0, 0.5])
    
    gmm_with_neg = query_with_negative_feedback(
        graph, "cup", failed_positions, n_explorations=1
    )
    prob_at_failed_neg = gmm_with_neg.evaluate([1.0, 0.5])
    
    # 验证指标
    reduction_rate = (prob_at_failed - prob_at_failed_neg) / (prob_at_failed + 1e-8)
    assert reduction_rate > 0.4, f"负向惩罚不足：{reduction_rate}"
    
    print(f"✅ 负向惩罚有效！在失败位置的概率下降 {reduction_rate*100:.1f}%")
```
